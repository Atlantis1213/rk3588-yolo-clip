cmake_minimum_required(VERSION 3.16)
project(yolo_clip_system_demo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)
option(BUILD_HARDWARE_DEMO "Build RKNN/RGA demo" ON)
option(ENABLE_SANITIZERS "ASan and UBSan for portable tests" OFF)
option(ENABLE_GPROF "Instrument normalization benchmark with gprof" OFF)
find_package(Threads REQUIRED)
include(CTest)

add_library(demo_headers INTERFACE)
target_include_directories(demo_headers INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(demo_headers INTERFACE -Wall -Wextra -fno-omit-frame-pointer)

if(BUILD_HARDWARE_DEMO)
  find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs videoio)
  find_path(RGA_INCLUDE_DIR im2d.h PATHS /usr/include/rga REQUIRED)
  find_path(RKNN_INCLUDE_DIR rknn_api.h PATHS /userdata/yolo_clip_bench /usr/include REQUIRED)
  find_library(RGA_LIBRARY rga REQUIRED)
  find_library(RKNN_LIBRARY rknnrt REQUIRED)
  add_library(demo_pipeline STATIC src/pipeline.cpp)
  target_include_directories(demo_pipeline PUBLIC ${RGA_INCLUDE_DIR} ${RKNN_INCLUDE_DIR} ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(demo_pipeline PUBLIC demo_headers Threads::Threads ${OpenCV_LIBS} ${RGA_LIBRARY} ${RKNN_LIBRARY})
  add_executable(yolo_clip_system_demo src/main.cpp)
  target_link_libraries(yolo_clip_system_demo PRIVATE demo_pipeline)
  add_executable(model_inventory tools/model_inventory.cpp)
  target_include_directories(model_inventory PRIVATE ${RKNN_INCLUDE_DIR})
  target_link_libraries(model_inventory PRIVATE demo_headers ${RKNN_LIBRARY})
  add_executable(bench_clip_concurrency tools/bench_clip_concurrency.cpp)
  target_link_libraries(bench_clip_concurrency PRIVATE demo_pipeline)
  add_executable(dump_clip_embeddings tools/dump_clip_embeddings.cpp)
  target_link_libraries(dump_clip_embeddings PRIVATE demo_pipeline)
  add_executable(prepare_real_clip_inputs tools/prepare_real_clip_inputs.cpp)
  target_link_libraries(prepare_real_clip_inputs PRIVATE demo_pipeline)
  add_executable(dump_frozen_clip_inputs tools/dump_frozen_clip_inputs.cpp)
  target_link_libraries(dump_frozen_clip_inputs PRIVATE demo_pipeline)
  add_executable(freeze_nonyolo_clip_inputs tools/freeze_nonyolo_clip_inputs.cpp)
  target_link_libraries(freeze_nonyolo_clip_inputs PRIVATE demo_pipeline)
  add_executable(probe_yolo_input_contract tools/probe_yolo_input_contract.cpp)
  target_link_libraries(probe_yolo_input_contract PRIVATE demo_pipeline)
  add_executable(bench_clip_batch tools/bench_clip_batch.cpp)
  target_link_libraries(bench_clip_batch PRIVATE demo_pipeline)
  add_executable(probe_quant_io tools/probe_quant_io.cpp)
  target_link_libraries(probe_quant_io PRIVATE demo_pipeline)
  add_executable(probe_quant_input tools/probe_quant_input.cpp)
  target_link_libraries(probe_quant_input PRIVATE demo_pipeline)
  add_executable(probe_input_layout tools/probe_input_layout.cpp)
  target_include_directories(probe_input_layout PRIVATE ${RKNN_INCLUDE_DIR} ${OpenCV_INCLUDE_DIRS} ${RGA_INCLUDE_DIR})
  target_link_libraries(probe_input_layout PRIVATE demo_headers ${RKNN_LIBRARY})
  add_executable(dump_prefix_outputs tools/dump_prefix_outputs.cpp)
  target_include_directories(dump_prefix_outputs PRIVATE ${RKNN_INCLUDE_DIR} ${OpenCV_INCLUDE_DIRS} ${RGA_INCLUDE_DIR})
  target_link_libraries(dump_prefix_outputs PRIVATE demo_headers ${RKNN_LIBRARY})
  add_executable(dump_standalone_sdpa tools/dump_standalone_sdpa.cpp)
  target_include_directories(dump_standalone_sdpa PRIVATE ${RKNN_INCLUDE_DIR} ${OpenCV_INCLUDE_DIRS} ${RGA_INCLUDE_DIR})
  target_link_libraries(dump_standalone_sdpa PRIVATE demo_headers ${RKNN_LIBRARY})
  if(ENABLE_GPROF)
    target_compile_options(demo_pipeline PRIVATE -pg)
    target_compile_options(yolo_clip_system_demo PRIVATE -pg)
    target_link_options(yolo_clip_system_demo PRIVATE -pg)
  endif()
endif()

add_executable(bench_normalization tests/bench_normalization.cpp)
target_link_libraries(bench_normalization PRIVATE demo_headers)
if(ENABLE_GPROF)
  target_compile_options(bench_normalization PRIVATE -pg)
  target_link_options(bench_normalization PRIVATE -pg)
endif()

if(BUILD_TESTING)
  foreach(test_name IN ITEMS test_normalization test_queue_pool test_npu_arbiter)
    add_executable(${test_name} tests/${test_name}.cpp)
    target_link_libraries(${test_name} PRIVATE demo_headers Threads::Threads)
    if(ENABLE_SANITIZERS)
      target_compile_options(${test_name} PRIVATE -fsanitize=address,undefined)
      target_link_options(${test_name} PRIVATE -fsanitize=address,undefined)
    endif()
    add_test(NAME ${test_name} COMMAND ${test_name})
    set_tests_properties(${test_name} PROPERTIES TIMEOUT 30)
  endforeach()
  # Deliberately reproduces the original shutdown ownership bug; not a passing CTest.
  add_executable(legacy_slot_leak_repro tests/legacy_slot_leak_repro.cpp)
  target_link_libraries(legacy_slot_leak_repro PRIVATE demo_headers Threads::Threads)
endif()
